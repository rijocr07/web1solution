Experiment 3 - Converter

Time.java

package Converter;

import java.util.Scanner;

public class Time {

    int hours;
    int minutes;
    int seconds;

    public void timeConverter() {

        Scanner sc = new Scanner(System.in);

        System.out.println("Enter the time in hours");
        hours = sc.nextInt();

        minutes = hours * 60;
        seconds = hours * 3600;

        System.out.println("hours = " + hours);
        System.out.println("minutes = " + minutes);
        System.out.println("Seconds = " + seconds);
    }
}

package convertermain;

import Converter.*;
import java.util.Scanner;

public class Convertermain {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        Curr c = new Curr();
        Dist d = new Dist();
        Time t = new Time();

        int choice;
        int ch;

        do {

            System.out.println("\n1. Currency");
            System.out.println("2. Distance");
            System.out.println("3. Time");

            System.out.println("Enter your choice");
            choice = sc.nextInt();

            switch (choice) {

                case 1:
                    c.RupeesConverter();
                    break;

                case 2:
                    d.Converterkm();
                    break;

                case 3:
                    t.timeConverter();
                    break;

                default:
                    System.out.println("Invalid choice");
            }

            System.out.println("Enter 1 for continue");
            ch = sc.nextInt();

        } while (ch == 1);

        sc.close();
    }
}

Experiment 5 - Linear Search Using Object and Class

LinearSearchMain.java

import java.util.Scanner;

class LinearSearch {

    int[] arr;
    int n;

    void getData() {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the number of terms : ");
        n = sc.nextInt();

        arr = new int[n];

        for (int i = 0; i < n; i++) {
            System.out.print("Enter the " + (i + 1) + " element: ");
            arr[i] = sc.nextInt();
        }
    }

    void search() {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the element to search : ");
        int key = sc.nextInt();

        for (int i = 0; i < n; i++) {

            if (arr[i] == key) {
                System.out.println("The element found at " + (i + 1) + " position");
                return;
            }
        }

        System.out.println("The element not found");
    }
}

public class LinearSearchMain {

    public static void main(String[] args) {

        LinearSearch obj = new LinearSearch();

        obj.getData();
        obj.search();
    }
}
